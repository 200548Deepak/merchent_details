( for virtual environment )
source venv/bin/activate

(to move to your merchent_details folder )
cd merchent_details

(to run in background)
nohup /home/ubuntu/merchent_details/venv/bin/python3 fetch2.py > fetch2.log 2>&1 &

(To check weather the fetch.py is running)
ps aux | grep "[f]etch2.py"
or
top

(To stop the process)
pkill -f fetch2.py

(To ome out of virtual environment)
deactivate
